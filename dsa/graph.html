🧱 2 · Setup

Use an adjacency list: graph[u] = the neighbours of u. It takes O(V + E) space. Use a matrix only when you need an O(1) "is there an edge?" check on a small, dense graph.

const DIRS = [[1, 0], [-1, 0], [0, 1], [0, -1]];   // down, up, right, left

// Edge list → adjacency list. Create every node first so isolated nodes exist.
function buildGraph(n: number, edges: number[][], directed: boolean): number[][] {
  const graph: number[][] = Array.from({ length: n }, () => []);
  for (const [u, v] of edges) {
    graph[u].push(v);
    if (!directed) graph[v].push(u);   // undirected = add both ways
  }
  return graph;
}
// Weighted: graph[u].push([v, w])
⚠️ Edge direction "To take course 1 you must first take 0" means [1, 0], and the edge goes 0 → 1. Draw two nodes before you code; getting the direction wrong gives silent wrong answers.

A heap for Dijkstra and Kruskal

JavaScript has no built-in heap. LeetCode's JS runtime has MinPriorityQueue, but an interview doc does not, so be able to type this in 3 minutes.

// Min-heap of arrays, ordered by item[0]. push / pop are O(log n).
class MinHeap {
  private items: number[][] = [];

  get size(): number { return this.items.length; }

  push(item: number[]): void {
    const a = this.items;
    a.push(item);
    let i = a.length - 1;
    while (i > 0) {                          // bubble up while smaller than parent
      const parent = Math.floor((i - 1) / 2);
      if (a[parent][0] <= a[i][0]) break;
      [a[parent], a[i]] = [a[i], a[parent]];
      i = parent;
    }
  }

  pop(): number[] {
    const a = this.items;
    const top = a[0];
    const last = a.pop()!;
    if (a.length > 0) {
      a[0] = last;                           // move last to root, then sink it
      let i = 0;
      while (true) {
        const left = 2 * i + 1, right = 2 * i + 2;
        let smallest = i;
        if (left < a.length && a[left][0] < a[smallest][0]) smallest = left;
        if (right < a.length && a[right][0] < a[smallest][0]) smallest = right;
        if (smallest === i) break;
        [a[smallest], a[i]] = [a[i], a[smallest]];
        i = smallest;
      }
    }
    return top;
  }
}

🌊 3 · BFS: fewest steps

"Explore in rings. The first time you reach a node is the shortest way there."

Only true when every edge costs the same. Process the graph one level at a time: build a next list from the current queue, then swap. The level number is the distance. This avoids queue.shift(), which is O(n).

// 994 Rotting Oranges — multi-source BFS. One level = one minute.
function orangesRotting(grid: number[][]): number {
  const rows = grid.length, cols = grid[0].length;
  let queue: number[][] = [];
  let fresh = 0;
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      if (grid[r][c] === 2) queue.push([r, c]);   // ALL sources start together
      if (grid[r][c] === 1) fresh++;
    }
  }

  let minutes = 0;
  while (queue.length > 0 && fresh > 0) {
    const next: number[][] = [];                   // the next level
    for (const [r, c] of queue) {
      for (const [dr, dc] of DIRS) {
        const nr = r + dr, nc = c + dc;
        if (nr < 0 || nc < 0 || nr >= rows || nc >= cols) continue;
        if (grid[nr][nc] !== 1) continue;
        grid[nr][nc] = 2;                          // mark when ADDED, not when popped
        fresh--;
        next.push([nr, nc]);
      }
    }
    queue = next;
    minutes++;
  }
  return fresh === 0 ? minutes : -1;
}
🔑 Mark when you ADD Mark a node visited when you push it, not when you pop it. Otherwise the same node gets pushed by many neighbours and the queue blows up.

Implicit graph: generate the neighbours

// 127 Word Ladder — BFS over words. Neighbours are generated, not stored.
function ladderLength(beginWord: string, endWord: string, wordList: string[]): number {
  const words = new Set(wordList);
  if (!words.has(endWord)) return 0;

  const seen = new Set([beginWord]);
  let queue = [beginWord];
  let length = 1;                                  // counts words, not edges
  while (queue.length > 0) {
    const next: string[] = [];
    for (const word of queue) {
      if (word === endWord) return length;
      for (let i = 0; i < word.length; i++) {
        for (const ch of 'abcdefghijklmnopqrstuvwxyz') {
          const candidate = word.slice(0, i) + ch + word.slice(i + 1);
          if (words.has(candidate) && !seen.has(candidate)) {
            seen.add(candidate);
            next.push(candidate);
          }
        }
      }
    }
    queue = next;
    length++;
  }
  return 0;
}

🌀 4 · DFS: islands, copies, cycles

"Go as deep as you can, then back up."

Use it for anything about structure: components, copying, cycles, colouring, and searching all paths. Recursion is the cleanest; switch to an explicit stack only if depth can pass about 10,000.

// 200 Number of Islands — outer loop counts, DFS sinks the whole island.
function numIslands(grid: string[][]): number {
  const rows = grid.length, cols = grid[0].length;

  function sink(r: number, c: number): void {
    if (r < 0 || c < 0 || r >= rows || c >= cols || grid[r][c] !== '1') return;
    grid[r][c] = '0';                              // mark visited in place
    for (const [dr, dc] of DIRS) sink(r + dr, c + dc);
  }

  let islands = 0;
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      if (grid[r][c] === '1') {
        islands++;                                 // new island found
        sink(r, c);
      }
    }
  }
  return islands;
}

Clone a graph

// 133 Clone Graph — map old node → copy. The map is also the visited set.
function cloneGraph(node: _Node | null): _Node | null {
  const copies = new Map<_Node, _Node>();

  function clone(original: _Node): _Node {
    if (copies.has(original)) return copies.get(original)!;
    const copy = new _Node(original.val);
    copies.set(original, copy);                    // save BEFORE recursing (cycles!)
    for (const nb of original.neighbors) copy.neighbors.push(clone(nb));
    return copy;
  }

  return node ? clone(node) : null;
}

Cycle in a directed graph

// Directed cycle — 3 states. Meeting a node that is still "in progress" = cycle.
function hasCycle(graph: number[][]): boolean {
  const state = new Array(graph.length).fill(0);   // 0 new · 1 in progress · 2 done

  function dfs(node: number): boolean {
    state[node] = 1;
    for (const next of graph[node]) {
      if (state[next] === 1) return true;          // back to our own path
      if (state[next] === 0 && dfs(next)) return true;
    }
    state[node] = 2;                               // fully explored, no cycle through it
    return false;
  }

  for (let i = 0; i < graph.length; i++) {
    if (state[i] === 0 && dfs(i)) return true;
  }
  return false;
}
⚠️ Why not one visited set? In a directed graph, A→B, A→C, B→D, C→D reaches D twice with no cycle. A plain visited check would wrongly report one. Only a node that is still in progress (on the current path) proves a cycle.

Undirected cycle: use Union-Find (§6): an edge whose two ends are already connected closes a cycle. 261 Valid Tree🔒: exactly n − 1 edges and everything connected.

Two groups: bipartite

// 785 Is Graph Bipartite — paint neighbours the opposite colour.
function isBipartite(graph: number[][]): boolean {
  const color = new Array(graph.length).fill(-1);  // -1 = not painted

  function paint(node: number, c: number): boolean {
    color[node] = c;
    for (const next of graph[node]) {
      if (color[next] === c) return false;         // same colour on both ends
      if (color[next] === -1 && !paint(next, 1 - c)) return false;
    }
    return true;
  }

  for (let i = 0; i < graph.length; i++) {         // graph may be disconnected
    if (color[i] === -1 && !paint(i, 0)) return false;
  }
  return true;
}

Weighted DFS: Evaluate Division

// 399 Evaluate Division — a/b = 2 means edge a→b (2) and b→a (1/2).
function calcEquation(equations: string[][], values: number[], queries: string[][]): number[] {
  const graph = new Map<string, [string, number][]>();
  function addEdge(from: string, to: string, weight: number): void {
    if (!graph.has(from)) graph.set(from, []);
    graph.get(from)!.push([to, weight]);
  }
  equations.forEach(([a, b], i) => {
    addEdge(a, b, values[i]);
    addEdge(b, a, 1 / values[i]);
  });

  // Multiply weights along any path from node to target.
  function dfs(node: string, target: string, seen: Set<string>): number {
    if (node === target) return 1;
    seen.add(node);
    for (const [next, weight] of graph.get(node)!) {
      if (seen.has(next)) continue;
      const rest = dfs(next, target, seen);
      if (rest !== -1) return weight * rest;
    }
    return -1;
  }

  return queries.map(([a, b]) => (graph.has(a) && graph.has(b) ? dfs(a, b, new Set()) : -1));
}

📊 5 · Topological sort: order with prerequisites

"Repeatedly take a task that nothing is waiting on."

Count each node's incoming edges (indegree). Start with every node at 0. Taking a node lowers its neighbours' counts, and any that reach 0 join the queue. If you finish with fewer than n nodes, the rest are stuck in a cycle. That one check also answers 207 "can you finish?"

// 210 Course Schedule II — Kahn's: keep taking nodes with no incoming edges.
function findOrder(numCourses: number, prerequisites: number[][]): number[] {
  const graph: number[][] = Array.from({ length: numCourses }, () => []);
  const indegree = new Array(numCourses).fill(0);
  for (const [course, pre] of prerequisites) {
    graph[pre].push(course);                       // pre → course
    indegree[course]++;
  }

  const order: number[] = [];                      // also used as the queue
  for (let i = 0; i < numCourses; i++) {
    if (indegree[i] === 0) order.push(i);
  }
  for (let i = 0; i < order.length; i++) {         // order grows as we read it
    for (const next of graph[order[i]]) {
      indegree[next]--;
      if (indegree[next] === 0) order.push(next);
    }
  }
  return order.length === numCourses ? order : []; // stuck before n → cycle
}
// 269 Alien Dictionary — compare neighbours, first different letter = one edge.
function alienOrder(words: string[]): string {
  const graph = new Map<string, Set<string>>();
  const indegree = new Map<string, number>();
  for (const word of words) {                      // every letter is a node
    for (const ch of word) {
      graph.set(ch, graph.get(ch) ?? new Set());
      indegree.set(ch, indegree.get(ch) ?? 0);
    }
  }

  for (let i = 0; i + 1 < words.length; i++) {
    const a = words[i], b = words[i + 1];
    if (a.length > b.length && a.startsWith(b)) return '';   // "abc" before "ab" = invalid
    for (let j = 0; j < Math.min(a.length, b.length); j++) {
      if (a[j] !== b[j]) {
        if (!graph.get(a[j])!.has(b[j])) {         // skip duplicate edges
          graph.get(a[j])!.add(b[j]);
          indegree.set(b[j], indegree.get(b[j])! + 1);
        }
        break;                                     // only the FIRST difference counts
      }
    }
  }

  const order: string[] = [];                      // Kahn's, same as 210
  for (const [ch, deg] of indegree) if (deg === 0) order.push(ch);
  for (let i = 0; i < order.length; i++) {
    for (const next of graph.get(order[i])!) {
      indegree.set(next, indegree.get(next)! - 1);
      if (indegree.get(next) === 0) order.push(next);
    }
  }
  return order.length === indegree.size ? order.join('') : '';
}

🔗 6 · Union-Find: who is in which group

"Every group has a root. Two nodes are connected if they have the same root."

Best when edges arrive one at a time and you keep asking "connected yet?" With path compression and union by size, each operation is nearly O(1). It works on undirected graphs only.

class UnionFind {
  parent: number[];
  size: number[];
  count: number;                                   // number of separate groups

  constructor(n: number) {
    this.parent = Array.from({ length: n }, (_, i) => i);   // everyone is their own root
    this.size = new Array(n).fill(1);
    this.count = n;
  }

  find(x: number): number {
    if (this.parent[x] !== x) this.parent[x] = this.find(this.parent[x]);  // path compression
    return this.parent[x];
  }

  // Returns false if a and b were already connected (this edge would make a cycle).
  union(a: number, b: number): boolean {
    let rootA = this.find(a), rootB = this.find(b);
    if (rootA === rootB) return false;
    if (this.size[rootA] < this.size[rootB]) [rootA, rootB] = [rootB, rootA];
    this.parent[rootB] = rootA;                    // hang the smaller tree under the bigger
    this.size[rootA] += this.size[rootB];
    this.count--;
    return true;
  }
}
// 684 Redundant Connection — the first edge whose ends are already connected.
function findRedundantConnection(edges: number[][]): number[] {
  const uf = new UnionFind(edges.length + 1);
  for (const [a, b] of edges) {
    if (!uf.union(a, b)) return [a, b];
  }
  return [];
}
// 721 Accounts Merge — accounts are nodes; a shared email joins them.
function accountsMerge(accounts: string[][]): string[][] {
  const uf = new UnionFind(accounts.length);
  const owner = new Map<string, number>();         // email → first account that had it
  accounts.forEach((account, i) => {
    for (const email of account.slice(1)) {
      if (owner.has(email)) uf.union(i, owner.get(email)!);
      else owner.set(email, i);
    }
  });

  const groups = new Map<number, string[]>();      // root account → all its emails
  for (const [email, i] of owner) {
    const root = uf.find(i);
    if (!groups.has(root)) groups.set(root, []);
    groups.get(root)!.push(email);
  }
  return [...groups].map(([root, emails]) => [accounts[root][0], ...emails.sort()]);
}

🛣️ 7 · Weighted shortest path

"BFS, but always expand the cheapest node next."

Dijkstra works when every weight is ≥ 0. JS heaps can't lower a key in place, so push a new entry and skip stale ones when they pop. Time is O(E log V).

// 743 Network Delay Time — Dijkstra: always expand the closest node next.
function networkDelayTime(times: number[][], n: number, k: number): number {
  const graph: number[][][] = Array.from({ length: n + 1 }, () => []);
  for (const [u, v, w] of times) graph[u].push([v, w]);

  const dist = new Array(n + 1).fill(Infinity);
  dist[k] = 0;
  const heap = new MinHeap();
  heap.push([0, k]);                               // [distance, node]

  while (heap.size > 0) {
    const [d, node] = heap.pop();
    if (d > dist[node]) continue;                  // stale: a shorter path already won
    for (const [next, w] of graph[node]) {
      if (d + w < dist[next]) {
        dist[next] = d + w;
        heap.push([dist[next], next]);
      }
    }
  }

  const slowest = Math.max(...dist.slice(1));      // nodes are 1..n
  return slowest === Infinity ? -1 : slowest;
}

The variation interviewers love: cost is a max, not a sum

// 1631 Path With Minimum Effort — grid Dijkstra where path cost = the MAX step.
function minimumEffortPath(heights: number[][]): number {
  const rows = heights.length, cols = heights[0].length;
  const effort = Array.from({ length: rows }, () => new Array(cols).fill(Infinity));
  effort[0][0] = 0;
  const heap = new MinHeap();
  heap.push([0, 0, 0]);                            // [effort, row, col]

  while (heap.size > 0) {
    const [e, r, c] = heap.pop();
    if (r === rows - 1 && c === cols - 1) return e;  // first time target pops = answer
    if (e > effort[r][c]) continue;
    for (const [dr, dc] of DIRS) {
      const nr = r + dr, nc = c + dc;
      if (nr < 0 || nc < 0 || nr >= rows || nc >= cols) continue;
      const next = Math.max(e, Math.abs(heights[nr][nc] - heights[r][c]));  // ⭐ max, not +
      if (next < effort[nr][nc]) {
        effort[nr][nc] = next;
        heap.push([next, nr, nc]);
      }
    }
  }
  return 0;
}

At most K stops: Bellman-Ford

// 787 Cheapest Flights Within K Stops — Bellman-Ford, one round per edge used.
function findCheapestPrice(n: number, flights: number[][], src: number, dst: number, k: number): number {
  let cost = new Array(n).fill(Infinity);
  cost[src] = 0;
  for (let round = 0; round <= k; round++) {       // k stops = k + 1 flights
    const next = [...cost];                        // copy: read last round, write this one
    for (const [from, to, price] of flights) {
      if (cost[from] + price < next[to]) next[to] = cost[from] + price;
    }
    cost = next;
  }
  return cost[dst] === Infinity ? -1 : cost[dst];
}
⚠️ Negative weights Dijkstra assumes a popped node is final, and a negative edge can break that later. With negative weights use Bellman-Ford: relax every edge V − 1 times, O(V·E). If a V-th pass still improves something, there's a negative cycle.

All pairs (1334, n ≤ 100): Floyd-Warshall, three loops with k outermost: dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j]). O(V³) time, O(V²) space.

🎤 Interview lens Probed: why the stale check exists, why there are no negative weights, and why not just BFS (weights differ). Escalation: 743 → 1631 (max) → 787 (K stops) → 1976 (count paths) → return the path.

🌳 8 · Minimum spanning tree

"Take the cheapest edge that doesn't make a cycle. Repeat."

That's Kruskal: sort the edges, then use Union-Find to skip any edge that joins two nodes already connected. A tree on n nodes has n − 1 edges, so stop once everything is connected.

// 1584 Min Cost to Connect All Points — Kruskal: cheapest edges first, skip cycles.
function minCostConnectPoints(points: number[][]): number {
  const n = points.length;
  const edges: number[][] = [];
  for (let i = 0; i < n; i++) {
    for (let j = i + 1; j < n; j++) {
      const dist = Math.abs(points[i][0] - points[j][0]) + Math.abs(points[i][1] - points[j][1]);
      edges.push([dist, i, j]);
    }
  }
  edges.sort((a, b) => a[0] - b[0]);

  const uf = new UnionFind(n);
  let total = 0;
  for (const [dist, i, j] of edges) {
    if (uf.union(i, j)) total += dist;             // joins two groups → keep the edge
    if (uf.count === 1) break;                     // everything connected
  }
  return total;
}
  • Prim grows one tree from a node, always taking the cheapest edge leaving it. It's Dijkstra's code with w in place of d + w. On 1584 (every pair is an edge) an array-based Prim is O(n²) and beats sorting n² edges.
  • 1168🔒 Optimize Water Distribution: add a fake node 0 and give each house an edge to it costing its well price. Now it's a plain MST.

Use every edge once: Hierholzer

// 332 Reconstruct Itinerary — Hierholzer: add an airport only when it has no tickets left.
function findItinerary(tickets: string[][]): string[] {
  const graph = new Map<string, string[]>();
  for (const [from, to] of tickets) {
    if (!graph.has(from)) graph.set(from, []);
    graph.get(from)!.push(to);
  }
  for (const dests of graph.values()) dests.sort().reverse();  // pop() gives the smallest

  const route: string[] = [];
  function dfs(airport: string): void {
    const dests = graph.get(airport) ?? [];
    while (dests.length > 0) dfs(dests.pop()!);    // use each ticket once
    route.push(airport);                           // post-order
  }
  dfs('JFK');
  return route.reverse();
}

Visiting every edge once (Euler) is O(E). Visiting every node once (Hamiltonian) is NP-hard, so don't try to code it. In a directed graph, a path exists if every node has in = out, except possibly one start (out = in + 1) and one end.

Bridges: critical connections

// 1192 Critical Connections — an edge is a bridge if the far side can't climb back above it.
function criticalConnections(n: number, connections: number[][]): number[][] {
  const graph = buildGraph(n, connections, false);
  const disc = new Array(n).fill(-1);              // when we first reached the node
  const low = new Array(n).fill(0);                // earliest disc reachable from its subtree
  const bridges: number[][] = [];
  let time = 0;

  function dfs(node: number, parent: number): void {
    disc[node] = low[node] = time++;
    for (const next of graph[node]) {
      if (next === parent) continue;
      if (disc[next] === -1) {
        dfs(next, node);
        low[node] = Math.min(low[node], low[next]);
        if (low[next] > disc[node]) bridges.push([node, next]);
      } else {
        low[node] = Math.min(low[node], disc[next]);  // back edge
      }
    }
  }
  dfs(0, -1);
  return bridges;
}

Articulation points (critical nodes) use the same code with low[next] >= disc[node], plus a rule that the DFS root counts only if it has 2 or more children.

Visit all nodes: BFS over (node, bitmask)

// 847 Shortest Path Visiting All Nodes — BFS where a state is (node, set of visited nodes).
function shortestPathLength(graph: number[][]): number {
  const n = graph.length;
  const all = (1 << n) - 1;                        // every bit set = every node visited
  const seen = Array.from({ length: n }, () => new Array(1 << n).fill(false));
  let queue: number[][] = [];
  for (let i = 0; i < n; i++) {                    // may start anywhere
    queue.push([i, 1 << i]);
    seen[i][1 << i] = true;
  }

  let steps = 0;
  while (queue.length > 0) {
    const next: number[][] = [];
    for (const [node, mask] of queue) {
      if (mask === all) return steps;
      for (const nb of graph[node]) {
        const nextMask = mask | (1 << nb);
        if (!seen[nb][nextMask]) {
          seen[nb][nextMask] = true;
          next.push([nb, nextMask]);
        }
      }
    }
    queue = next;
    steps++;
  }
  return -1;
}
