MinHeap Implementation

class MinHeap<T> {
  private heap: T[] = [];
  constructor(private compareFn: (a: T, b: T) => number) {}

  push(val: T): void {
    this.heap.push(val);
    this._bubbleUp(this.heap.length - 1);
  }

  pop(): T | undefined {
    if (this.heap.length === 0) return undefined;
    if (this.heap.length === 1) return this.heap.pop();

    const min = this.heap[0];
    this.heap[0] = this.heap.pop()!;
    this._bubbleDown(0);
    return min;
  }

  isEmpty(): boolean {
    return this.heap.length === 0;
  }
  size(): number {
    return this.heap.length;
  }

  private _bubbleUp(idx: number): void {
    while (idx > 0) {
      const parentIdx = Math.floor((idx - 1) / 2);
      if (this.compareFn(this.heap[idx], this.heap[parentIdx]) >= 0) break;
      [this.heap[idx], this.heap[parentIdx]] = [this.heap[parentIdx], this.heap[idx]];
      idx = parentIdx;
    }
  }

  private _bubbleDown(idx: number): void {
    while (true) {
      let smallest = idx;
      const left = 2 * idx + 1;
      const right = 2 * idx + 2;

      if (left < this.heap.length && this.compareFn(this.heap[left], this.heap[smallest]) < 0) {
        smallest = left;
      }
      if (right < this.heap.length && this.compareFn(this.heap[right], this.heap[smallest]) < 0) {
        smallest = right;
      }

      if (smallest === idx) break;
      [this.heap[idx], this.heap[smallest]] = [this.heap[smallest], this.heap[idx]];
      idx = smallest;
    }
  }
}

Union-Find with Path Compression + Union by Rank

class UnionFind {
  parent: number[];
  rank: number[];

  constructor(n: number) {
    this.parent = Array.from({ length: n }, (_, i) => i);
    this.rank = new Array(n).fill(0);
  }

  find(x: number): number {
    if (this.parent[x] !== x) {
      // Path compression: point directly to root
      this.parent[x] = this.find(this.parent[x]);
    }
    return this.parent[x];
  }

  union(x: number, y: number): boolean {
    const rootX = this.find(x);
    const rootY = this.find(y);

    if (rootX === rootY) return false; // already in same set

    // Union by rank: attach smaller tree under larger
    if (this.rank[rootX] < this.rank[rootY]) {
      this.parent[rootX] = rootY;
    } else if (this.rank[rootX] > this.rank[rootY]) {
      this.parent[rootY] = rootX;
    } else {
      this.parent[rootY] = rootX;
      this.rank[rootX]++;
    }
    return true; // successful union
  }

  isConnected(x: number, y: number): boolean {
    return this.find(x) === this.find(y);
  }
}

// ⭐ Why path compression + rank matters:
// Without: O(log n) per operation
// With both: O(α(n)) amortized ≈ O(1) for practical purposes
// Proof: Ackermann function inverse grows ABSURDLY slow

2.1 Standard BFS (Unweighted Graph)

function bfs(graph: Graph, start: number, n: number): void {
  const visited = new Set<number>([start]);
  const queue: number[] = [start];
  let head = 0; // ⭐ Use index pointer, NOT shift()

  while (head < queue.length) {
    const node = queue[head++];

    // ── PROCESS node HERE ──
    console.log(node);

    for (const neighbor of graph.get(node) ?? []) {
      if (!visited.has(neighbor)) {
        visited.add(neighbor); // ⭐ Mark on ENQUEUE, not dequeue
        queue.push(neighbor);
      }
    }
  }
}
// Time: O(V+E)   Space: O(V)
// Why index pointer? shift() is O(n), pointer is O(1)

2.2 BFS with Distance/Level Tracking

function bfsWithDist(graph: Graph, start: number, n: number): Map<number, number> {
  const dist = new Map<number, number>([[start, 0]]);
  const queue: number[] = [start];
  let head = 0;

  while (head < queue.length) {
    const node = queue[head++];
    const currDist = dist.get(node)!;

    for (const neighbor of graph.get(node) ?? []) {
      if (!dist.has(neighbor)) {
        dist.set(neighbor, currDist + 1);
        queue.push(neighbor);
      }
    }
  }
  return dist;
}
// Returns Map of shortest distances from start
// Use when: "shortest path" question on unweighted graph

2.3 Multi-Source BFS (LC 994, 286, 417)

function bfsMultiSource(grid: string[][], sources: [number, number][]): number[][] {
  const rows = grid.length,
    cols = grid[0].length;
  const dist: number[][] = Array.from({ length: rows }, () => new Array(cols).fill(-1));
  const queue: [number, number][] = [];

  // Seed all sources with distance 0
  for (const [r, c] of sources) {
    dist[r][c] = 0;
    queue.push([r, c]);
  }

  let head = 0;
  const dirs = [
    [0, 1],
    [0, -1],
    [1, 0],
    [-1, 0],
  ];

  while (head < queue.length) {
    const [r, c] = queue[head++];

    for (const [dr, dc] of dirs) {
      const nr = r + dr,
        nc = c + dc;
      if (nr >= 0 && nr < rows && nc >= 0 && nc < cols && dist[nr][nc] === -1) {
        dist[nr][nc] = dist[r][c] + 1;
        queue.push([nr, nc]);
      }
    }
  }

  return dist;
}
// Use when: "distance to nearest X" or "distance from all Xs"
// Examples: LC 994 (rotting oranges), LC 286 (walls and gates)

2.4 Bidirectional BFS (SDE3 Pattern) ⭐

function bidirectionalBFS(start: string, end: string, getNeighbors: (state: string) => string[]): number {
  if (start === end) return 0;

  const queueA = [start];
  const queueB = [end];
  const visitedA = new Set<string>([start]);
  const visitedB = new Set<string>([end]);
  let depth = 0;

  while (queueA.length > 0 && queueB.length > 0) {
    // Always expand the smaller frontier (optimization)
    if (queueA.length > queueB.length) {
      [queueA, queueB] = [queueB, queueA];
      [visitedA, visitedB] = [visitedB, visitedA];
    }

    const nextQueueA: string[] = [];

    for (const state of queueA) {
      for (const neighbor of getNeighbors(state)) {
        if (visitedB.has(neighbor)) {
          return depth + 1; // met in middle
        }
        if (!visitedA.has(neighbor)) {
          visitedA.add(neighbor);
          nextQueueA.push(neighbor);
        }
      }
    }

    queueA = nextQueueA;
    depth++;
  }

  return -1; // unreachable
}

// ⭐ Why this matters for SDE3:
// Normal BFS: explores O(b^d) nodes where b=branching factor, d=depth
// Bidirectional: explores O(b^(d/2) + b^(d/2)) = 2*b^(d/2)
// This is EXPONENTIALLY faster for large d
// Examples: LC 127 (Word Ladder), LC 815 (Bus Routes)

2.5 0-1 BFS (Deque) — Special Case

function bfs01(
  graph: Map<number, [number, number][]>, // [neighbor, weight (0 or 1)]
  start: number,
  n: number,
): number[] {
  const dist = new Array(n).fill(Infinity);
  dist[start] = 0;
  const deque: number[] = [start];
  let front = 0,
    back = 1; // ✅ Use index pointers, NOT shift/unshift!

  while (front < back) {
    const node = deque[front++]; // ✅ O(1) dequeue

    for (const [neighbor, weight] of graph.get(node) ?? []) {
      const newDist = dist[node] + weight;
      if (newDist < dist[neighbor]) {
        dist[neighbor] = newDist;

        if (weight === 0) {
          deque[--front] = neighbor; // ✅ O(1) prepend (high priority)
        } else {
          deque[back++] = neighbor; // ✅ O(1) append (low priority)
        }
      }
    }
  }
  return dist;
}

// Use when: weights are only 0 or 1
// Time: O(V+E) — NEVER use shift/unshift, they are O(n)!
// Classic example: grid where some moves are free, some costly

🌀 SECTION 3: DFS TEMPLATES (4 Variants)

3.1 DFS Recursive (Cleanest)

function dfs(graph: Graph, node: number, visited: Set<number>): void {
  visited.add(node);

  // ── PRE-ORDER PROCESS (before visiting children) ──
  console.log(`entering ${node}`);

  for (const neighbor of graph.get(node) ?? []) {
    if (!visited.has(neighbor)) {
      dfs(graph, neighbor, visited);
    }
  }

  // ── POST-ORDER PROCESS (after visiting all children) ──
  console.log(`leaving ${node}`);
}

// For all components (handles disconnected graphs):
function traverseAll(graph: Graph, n: number): void {
  const visited = new Set<number>();
  for (let i = 0; i < n; i++) {
    if (!visited.has(i)) {
      dfs(graph, i, visited);
    }
  }
}
// Time: O(V+E)   Space: O(V) call stack depth
// ⚠️ Stack overflow risk on very deep graphs (see iterative version)

3.2 DFS Iterative (Stack Overflow Safe)

function dfsIterative(graph: Graph, start: number): void {
  const visited = new Set<number>();
  const stack: number[] = [start];

  while (stack.length > 0) {
    const node = stack.pop()!;

    if (visited.has(node)) continue;
    visited.add(node);

    // ── PROCESS node HERE ──
    console.log(node);

    // Push neighbors in REVERSE order to maintain DFS order
    for (let i = (graph.get(node) ?? []).length - 1; i >= 0; i--) {
      const neighbor = (graph.get(node) ?? [])[i];
      if (!visited.has(neighbor)) {
        stack.push(neighbor);
      }
    }
  }
}
// Use when: recursive depth could exceed ~10000 nodes
// Maintains same DFS traversal order as recursive version

3.3 DFS on Grid (In-Place Marking)

function dfsGridInPlace(grid: string[][], r: number, c: number): void {
  const rows = grid.length,
    cols = grid[0].length;

  // Boundary + visited check in one condition
  if (r < 0 || r >= rows || c < 0 || c >= cols || grid[r][c] !== '1') {
    return;
  }

  grid[r][c] = '#'; // ⭐ Mark visited by modifying grid

  dfsGridInPlace(grid, r + 1, c);
  dfsGridInPlace(grid, r - 1, c);
  dfsGridInPlace(grid, r, c + 1);
  dfsGridInPlace(grid, r, c - 1);
}

// Use when: memory is tight, you can modify input
// Classic: LC 200 (Number of Islands)

3.4 DFS with Backtracking (All Paths)

function dfsBacktrack(
  graph: Graph,
  node: number,
  target: number,
  path: number[],
  visited: Set<number>, // ✅ Use Set, NOT path.includes()!
  allPaths: number[][],
): void {
  if (node === target) {
    allPaths.push([...path]);
    return;
  }

  for (const neighbor of graph.get(node) ?? []) {
    if (!visited.has(neighbor)) {
      // ✅ O(1) lookup
      visited.add(neighbor);
      path.push(neighbor);
      dfsBacktrack(graph, neighbor, target, path, visited, allPaths);
      path.pop(); // ✅ BACKTRACK
      visited.delete(neighbor);
    }
  }
}

// Use when: "find all paths", not just one path
// Time: O(path length * degree) per recursion, exponential worst case
// ⚠️ CRITICAL: Use Set for O(1) lookup, NOT path.includes() which is O(n)!

🛣️ SECTION 4: SHORTEST PATH ALGORITHMS

4.1 Dijkstra (Weighted, Non-Negative)

function dijkstra(graph: WeightedGraph, start: number, n: number): number[] {
  const dist = new Array(n).fill(Infinity);
  dist[start] = 0;

  const heap = new MinHeap<[number, number]>((a, b) => a[0] - b[0]);
  heap.push([0, start]); // [distance, node]

  while (!heap.isEmpty()) {
    const [d, node] = heap.pop()!;

    // ⭐ CRITICAL: stale entry check
    if (d > dist[node]) continue;

    for (const [neighbor, weight] of graph.get(node) ?? []) {
      const newDist = dist[node] + weight;
      if (newDist < dist[neighbor]) {
        dist[neighbor] = newDist;
        heap.push([newDist, neighbor]);
      }
    }
  }

  return dist;
}

// Time: O((V+E)logV)   Space: O(V)
// Use when: shortest path + all weights ≥ 0
// ❌ DON'T use with negative weights (use Bellman-Ford instead)

4.2 Bellman-Ford (Negative Weights OK)

function bellmanFord(
  n: number,
  edges: number[][], // [u, v, weight]
  start: number,
): number[] {
  const dist = new Array(n).fill(Infinity);
  dist[start] = 0;

  // Relax edges V-1 times
  for (let i = 0; i < n - 1; i++) {
    for (const [u, v, w] of edges) {
      if (dist[u] !== Infinity && dist[u] + w < dist[v]) {
        dist[v] = dist[u] + w;
      }
    }
  }

  // Check for negative cycles (optional)
  for (const [u, v, w] of edges) {
    if (dist[u] !== Infinity && dist[u] + w < dist[v]) {
      return []; // or throw error — negative cycle detected
    }
  }

  return dist;
}

// Time: O(V·E)   Space: O(V)
// Use when: negative edge weights exist (but no negative cycles)
// ❌ Much slower than Dijkstra, only use if necessary

4.3 Bellman-Ford with K-Stops Limit (LC 787)

function cheapestFlights(
  n: number,
  flights: number[][], // [u, v, price]
  src: number,
  dst: number,
  k: number, // AT MOST k stops allowed (problem constraint)
): number {
  // k stops = k+1 edges maximum
  // So we do k iterations (each iteration = one more edge)
  let dist = new Array(n).fill(Infinity);
  dist[src] = 0;

  for (let i = 0; i < k; i++) {
    // ✅ k iterations = k stops
    const nextDist = [...dist]; // MUST copy to prevent value bleed
    let updated = false;

    for (const [u, v, price] of flights) {
      if (dist[u] !== Infinity && dist[u] + price < nextDist[v]) {
        nextDist[v] = dist[u] + price;
        updated = true;
      }
    }

    dist = nextDist;
    if (!updated) break; // ✅ Early exit if no improvement
  }

  return dist[dst] === Infinity ? -1 : dist[dst];
}

// ✅ Time: O(k * E)  Space: O(n)
// Key: k stops = exactly k iterations, not k+1

4.4 Floyd-Warshall (All Pairs)

function floydWarshall(n: number, edges: number[][]): number[][] {
  const dist: number[][] = Array.from({ length: n }, (_, i) => new Array(n).fill(Infinity));

  // Initialize
  for (let i = 0; i < n; i++) dist[i][i] = 0;
  for (const [u, v, w] of edges) {
    dist[u][v] = Math.min(dist[u][v], w);
  }

  // For each intermediate node k
  for (let k = 0; k < n; k++) {
    for (let i = 0; i < n; i++) {
      for (let j = 0; j < n; j++) {
        dist[i][j] = Math.min(dist[i][j], dist[i][k] + dist[k][j]);
      }
    }
  }

  return dist;
}

// Time: O(V³)   Space: O(V²)
// Use when: need shortest paths between ALL pairs
// Only viable for small graphs (V ≤ 500)

🌳 SECTION 5: MINIMUM SPANNING TREE

5.1 Kruskal's Algorithm (Sort Edges)

function kruskal(n: number, edges: number[][]): number {
  // edges = [[u, v, weight], ...]
  edges.sort((a, b) => a[2] - b[2]); // Sort by weight

  const uf = new UnionFind(n);
  let mstCost = 0;
  let edgesAdded = 0;

  for (const [u, v, w] of edges) {
    if (uf.union(u, v)) {
      // No cycle formed
      mstCost += w;
      edgesAdded++;
      if (edgesAdded === n - 1) break; // MST complete
    }
  }

  return edgesAdded === n - 1 ? mstCost : -1; // -1 if disconnected
}

// Time: O(E log E)   Space: O(V)
// Use when: sparse graph, edges easily enumerable
// Why: Sorting dominates, Union-Find is nearly O(1)

5.2 Prim's Algorithm (Start from Vertex)

function prims(graph: WeightedGraph, n: number): number {
  const visited = new Set<number>();
  const heap = new MinHeap<[number, number]>((a, b) => a[0] - b[0]);

  // [weight, node]
  heap.push([0, 0]);
  let mstCost = 0;

  while (!heap.isEmpty() && visited.size < n) {
    const [weight, node] = heap.pop()!;

    if (visited.has(node)) continue;
    visited.add(node);
    mstCost += weight;

    for (const [neighbor, w] of graph.get(node) ?? []) {
      if (!visited.has(neighbor)) {
        heap.push([w, neighbor]);
      }
    }
  }

  return visited.size === n ? mstCost : -1; // -1 if disconnected
}

// Time: O((V+E)logV)   Space: O(V)
// Use when: dense graph or starting from specific vertex
// Why: fewer iterations typically than Kruskal on dense graphs

4️⃣A: A* SEARCH & HEURISTIC PATHFINDING (LC 1293, LC 1129)

// A* = Dijkstra + Heuristic (f = g + h)
// g(n) = actual cost from start, h(n) = estimated cost to goal
// Use when: Dijkstra is too slow and you have a heuristic

function aStar(grid: number[][], start: [number, number], goal: [number, number]): number {
  const rows = grid.length,
    cols = grid[0].length;
  const gCost = new Map<string, number>();
  const openSet = new MinHeap<[number, number, number]>((a, b) => a[0] - b[0]);

  const manhattan = (a: [number, number], b: [number, number]) => Math.abs(a[0] - b[0]) + Math.abs(a[1] - b[1]);
  const key = (r: number, c: number) => `${r},${c}`;

  gCost.set(key(start[0], start[1]), 0);
  const heuristic = manhattan(start, goal);
  openSet.push([heuristic, start[0], start[1]]);

  while (!openSet.isEmpty()) {
    const [f, r, c] = openSet.pop()!;
    const g = gCost.get(key(r, c))!;

    if (r === goal[0] && c === goal[1]) return g;

    for (const [dr, dc] of [
      [0, 1],
      [0, -1],
      [1, 0],
      [-1, 0],
    ]) {
      const nr = r + dr,
        nc = c + dc;
      if (nr < 0 || nr >= rows || nc < 0 || nc >= cols || grid[nr][nc] === 1) continue;

      const newG = g + 1;
      const neighborKey = key(nr, nc);

      if (!gCost.has(neighborKey) || newG < gCost.get(neighborKey)!) {
        gCost.set(neighborKey, newG);
        const newF = newG + manhattan([nr, nc], goal);
        openSet.push([newF, nr, nc]);
      }
    }
  }

  return -1;
}

// Time: O(E log V) with good heuristic, faster than Dijkstra
// Use when: grid problems with visible target (2D/3D games)

4️⃣B: IMPLICIT GRAPH GENERATION (LC 127 Word Ladder Optimized)

// Don't materialize neighbors — generate on the fly
// Saves memory for huge state spaces (10^9+ possible states)

interface ImplicitGraph<T> {
  neighbors: (state: T) => T[];
}

function bfsImplicit<T>(graph: ImplicitGraph<T>, start: T, goal: T): number {
  const visited = new Set<string>();
  const queue: T[] = [start];
  let steps = 0;
  let head = 0;

  const toString = (state: T) => JSON.stringify(state);
  visited.add(toString(start));

  while (head < queue.length) {
    const size = queue.length;

    for (let i = 0; i < size - head; i++) {
      const state = queue[head++];

      if (toString(state) === toString(goal)) {
        return steps;
      }

      for (const neighbor of graph.neighbors(state)) {
        const key = toString(neighbor);
        if (!visited.has(key)) {
          visited.add(key);
          queue.push(neighbor);
        }
      }
    }
    steps++;
  }

  return -1;
}

// Example: Word Ladder (LC 127)
// Don't enumerate all words, generate neighbors by changing 1 letter
// Memory: O(1) instead of O(word count)

function isValidTree(n: number, edges: number[][]): boolean {

  if (edges.length !== n - 1) return false;

  const graph: Graph = new Map();
  for (let i = 0; i < n; i++) graph.set(i, []);

  for (const [u, v] of edges) {
    graph.get(u)!.push(v);
    graph.get(v)!.push(u);
  }

    visited.add(node);
    for (const neighbor of graph.get(node) ?? []) {
      if (!visited.has(neighbor)) dfs(neighbor);
    }
  }

  dfs(0);
  return visited.size === n; // All connected?
}

4️⃣D: DAG-SPECIFIC DP OPTIMIZATIONS (LC 1991, 1203)

// On DAGs: topo sort + DP is O(V+E) guaranteed
// Much better than memoized DFS which can degrade

function longestPathDAG(n: number, edges: number[][]): number {
  const graph: Graph = new Map();
  const indegree = new Array(n).fill(0);

  for (let i = 0; i < n; i++) graph.set(i, []);
  for (const [u, v] of edges) {
    graph.get(u)!.push(v);
    indegree[v]++;
  }

  // Topological sort via Kahn's
  const queue: number[] = [];
  for (let i = 0; i < n; i++) {
    if (indegree[i] === 0) queue.push(i);
  }

  const topo: number[] = [];
  let head = 0;

  while (head < queue.length) {
    const node = queue[head++];
    topo.push(node);

    for (const neighbor of graph.get(node) ?? []) {
      indegree[neighbor]--;
      if (indegree[neighbor] === 0) queue.push(neighbor);
    }
  }

  // DP in topo order
  const dp = new Array(n).fill(1);

  for (const node of topo) {
    for (const neighbor of graph.get(node) ?? []) {
      dp[neighbor] = Math.max(dp[neighbor], dp[node] + 1);
    }
  }

  return Math.max(...dp);
}

// Time: O(V+E) guaranteed (vs memoized DFS which can be O(V²))
// Signal: recognize when DAG structure allows DP optimization

📊 SECTION 7: TOPOLOGICAL SORT

7.1 Kahn's Algorithm (BFS-based, safe for cycle detection)

function topologicalSortKahn(graph: Graph, n: number): number[] {
  const indegree = new Array(n).fill(0);

  // Calculate indegrees
  for (let i = 0; i < n; i++) {
    for (const neighbor of graph.get(i) ?? []) {
      indegree[neighbor]++;
    }
  }

  // Start with all nodes of indegree 0
  const queue: number[] = [];
  for (let i = 0; i < n; i++) {
    if (indegree[i] === 0) queue.push(i);
  }

  const result: number[] = [];
  let head = 0;

  while (head < queue.length) {
    const node = queue[head++];
    result.push(node);

    for (const neighbor of graph.get(node) ?? []) {
      indegree[neighbor]--;
      if (indegree[neighbor] === 0) {
        queue.push(neighbor);
      }
    }
  }

  // If not all nodes processed → cycle exists
  return result.length === n ? result : [];
}

// Time: O(V+E)   Space: O(V)
// Advantage: Can easily detect cycles (if result.length < n)

7.2 DFS-based Topological Sort + 3-Color DFS for Cycle Detection

function topologicalSortDFS(graph: Graph, n: number): number[] {
  // color: 0=white(unvisited), 1=gray(visiting), 2=black(done)
  const color = new Array(n).fill(0);
  const result: number[] = [];
  let hasCycle = false;

  function dfs(node: number): void {
    if (hasCycle) return;
    if (color[node] === 1) {
      hasCycle = true; // Back edge found = cycle
      return;
    }
    if (color[node] === 2) return;

    color[node] = 1; // Mark as visiting

    for (const neighbor of graph.get(node) ?? []) {
      dfs(neighbor);
    }

    color[node] = 2; // Mark as done
    result.push(node); // Add after all children processed (post-order)
  }

  for (let i = 0; i < n; i++) {
    if (color[i] === 0) {
      dfs(i);
    }
  }

  return hasCycle ? [] : result.reverse();
}

// ⭐ Key difference from Kahn's:
// - DFS adds nodes in POST-order, then reverses
// - 3-color DFS detects back edges (cycle indicator)
// - Useful for directed cycle detection

🔄 SECTION 8: CYCLE DETECTION

8.1 Undirected Graph (DFS + Parent Check)

function hasCycleUndirected(graph: Graph, n: number): boolean {
  const visited = new Set<number>();

  function dfs(node: number, parent: number): boolean {
    visited.add(node);

    for (const neighbor of graph.get(node) ?? []) {
      if (!visited.has(neighbor)) {
        if (dfs(neighbor, node)) return true;
      } else if (neighbor !== parent) {
        // Visited node that is NOT parent = back edge = cycle
        return true;
      }
    }

    return false;
  }

  for (let i = 0; i < n; i++) {
    if (!visited.has(i)) {
      if (dfs(i, -1)) return true;
    }
  }

  return false;
}

// ⚠️ Edge case: Multiple edges between same nodes
// If graph allows multi-edges, parent check breaks!
// Solution: track edge count or use different approach

8.2 Directed Graph (3-Color DFS)

function hasCycleDirected(graph: Graph, n: number): boolean {
  const color = new Array(n).fill(0); // 0=white, 1=gray, 2=black

  function dfs(node: number): boolean {
    color[node] = 1; // Mark as visiting

    for (const neighbor of graph.get(node) ?? []) {
      if (color[neighbor] === 1) {
        // Back edge to node still in this DFS = cycle
        return true;
      }
      if (color[neighbor] === 0) {
        if (dfs(neighbor)) return true;
      }
    }

    color[node] = 2; // Mark as done
    return false;
  }

  for (let i = 0; i < n; i++) {
    if (color[i] === 0) {
      if (dfs(i)) return true;
    }
  }

  return false;
}

// Why 3 colors?
// - White: not visited yet
// - Gray: in current DFS stack
// - Black: finished, safe to reference
// Back edge to GRAY = cycle (to a node in same path)

function hasCycleUnionFind(n: number, edges: number[][]): boolean {

}

🎨 SECTION 9: BIPARTITE & 2-COLORING

9.1 Check if Bipartite (LC 785)

function isBipartite(graph: number[][]): boolean {
  const n = graph.length;
  const color = new Array(n).fill(-1); // -1=unvisited, 0/1=colors

  for (let start = 0; start < n; start++) {
    if (color[start] !== -1) continue;

    // BFS from unvisited component
    const queue = [start];
    color[start] = 0;
    let head = 0;

    while (head < queue.length) {
      const node = queue[head++];

      for (const neighbor of graph[node]) {
        if (color[neighbor] === -1) {
          color[neighbor] = 1 - color[node]; // Flip color
          queue.push(neighbor);
        } else if (color[neighbor] === color[node]) {
          // Same color = not bipartite
          return false;
        }
      }
    }
  }

  return true;
}

// ⭐ Bipartite = 2-colorable = NO odd-length cycles
// Time: O(V+E)   Space: O(V)

// Kosaraju's Algorithm — O(V+E)
// Returns: array of SCCs (groups of mutually reachable nodes)
function kosaraju(graph: Graph, n: number): number[][] {
  // Step 1: DFS to get finish times
  const visited = new Set<number>();
  const finishStack: number[] = [];

  function dfs1(node: number): void {
    visited.add(node);
    for (const neighbor of graph.get(node) ?? []) {
      if (!visited.has(neighbor)) dfs1(neighbor);
    }
    finishStack.push(node); // Push AFTER visiting children
  }

  for (let i = 0; i < n; i++) {
    if (!visited.has(i)) dfs1(i);
  }

  // Step 2: Build reversed graph
  const revGraph: Graph = new Map();
  for (let i = 0; i < n; i++) revGraph.set(i, []);
  for (const [node, neighbors] of graph) {
    for (const neighbor of neighbors) {
      revGraph.get(neighbor)!.push(node);
    }
  }

  // Step 3: DFS on reversed graph in reverse finish order
  visited.clear();
  const sccs: number[][] = [];

  function dfs2(node: number, component: number[]): void {
    visited.add(node);
    component.push(node);
    for (const neighbor of revGraph.get(node) ?? []) {
      if (!visited.has(neighbor)) dfs2(neighbor, component);
    }
  }

  while (finishStack.length > 0) {
    const node = finishStack.pop()!;
    if (!visited.has(node)) {
      const component: number[] = [];
      dfs2(node, component);
      sccs.push(component);
    }
  }

  return sccs;
}

// Time: O(V+E)   Space: O(V)
// Returns: array of strongly connected components
// Use: LC 1192 (Critical Connections), directed graph analysis

🌉 SECTION 11: BRIDGES & ARTICULATION POINTS

11.1 Find Bridges (Critical Edges)

function findBridges(graph: Graph, n: number): number[][] {
  const disc = new Array(n).fill(-1);
  const low = new Array(n).fill(-1);
  const bridges: number[][] = [];
  let timer = 0;

  function dfs(node: number, parent: number): void {
    disc[node] = low[node] = timer++;

    for (const neighbor of graph.get(node) ?? []) {
      if (disc[neighbor] === -1) {
        // Tree edge
        dfs(neighbor, node);
        low[node] = Math.min(low[node], low[neighbor]);

        // Bridge condition: cannot reach ancestor from subtree
        if (low[neighbor] > disc[node]) {
          bridges.push([node, neighbor]);
        }
      } else if (neighbor !== parent) {
        // Back edge
        low[node] = Math.min(low[node], disc[neighbor]);
      }
    }
  }

  for (let i = 0; i < n; i++) {
    if (disc[i] === -1) dfs(i, -1);
  }

  return bridges;
}

// Bridge = edge whose removal disconnects graph
// Use: LC 1192 (Critical Connections in Network)

12.1 Word Ladder (LC 127) — State = String

function ladderLength(beginWord: string, endWord: string, wordList: string[]): number {
  const wordSet = new Set(wordList);
  if (!wordSet.has(endWord)) return 0;

  const queue: [string, number][] = [[beginWord, 1]];
  const visited = new Set<string>([beginWord]);
  let head = 0;

  while (head < queue.length) {
    const [word, steps] = queue[head++];

    for (let i = 0; i < word.length; i++) {
      for (let c = 97; c <= 122; c++) {
        // 'a' to 'z'
        const newWord = word.slice(0, i) + String.fromCharCode(c) + word.slice(i + 1);

        if (newWord === endWord) return steps + 1;

        if (wordSet.has(newWord) && !visited.has(newWord)) {
          visited.add(newWord);
          queue.push([newWord, steps + 1]);
        }
      }
    }
  }

  return 0;
}

// ⭐ KEY INSIGHT: Edge not explicitly given, generated on-the-fly
// BFS finds shortest path between states in implicit graph

12.2 Open the Lock (LC 752) — State = String + Visited States

function openLock(deadends: string[], target: string): number {
  if (target === '0000') return 0;

  const dead = new Set(deadends);
  if (dead.has('0000')) return -1;

  const queue: [string, number][] = [['0000', 0]];
  const visited = new Set<string>(['0000']);
  let head = 0;

  while (head < queue.length) {
    const [code, steps] = queue[head++];

    for (let i = 0; i < 4; i++) {
      for (const delta of [1, -1]) {
        const digit = (parseInt(code[i]) + delta + 10) % 10;
        const nextCode = code.slice(0, i) + digit + code.slice(i + 1);

        if (nextCode === target) return steps + 1;

        if (!dead.has(nextCode) && !visited.has(nextCode)) {
          visited.add(nextCode);
          queue.push([nextCode, steps + 1]);
        }
      }
    }
  }

  return -1;
}

// Graph structure:
// Nodes = combinations of lock (10^4 possible)
// Edges = ±1 twist on each dial
// BFS finds shortest path to target, avoiding dead ends

12.3 Multi-Dimensional State (LC 1293, 864)

// State = (row, col, keys_collected) — represented as tuple or bitmask

function shortestPathObstacles(grid: number[][], start: [number, number], end: [number, number], k: number): number {
  const rows = grid.length,
    cols = grid[0].length;
  // dist[r][c][eliminations_left] = min steps
  const dist: number[][][] = Array.from({ length: rows }, () =>
    Array.from({ length: cols }, () => new Array(k + 1).fill(Infinity)),
  );

  dist[start[0]][start[1]][k] = 0;
  const queue: [number, number, number, number][] = [[start[0], start[1], k, 0]]; // [r, c, elim_left, steps]

  while (queue.length > 0) {
    const [r, c, elim, steps] = queue.shift()!;

    if (r === end[0] && c === end[1]) return steps;

    for (const [dr, dc] of [
      [0, 1],
      [0, -1],
      [1, 0],
      [-1, 0],
    ]) {
      const nr = r + dr,
        nc = c + dc;
      if (nr < 0 || nr >= rows || nc < 0 || nc >= cols) continue;

      const newElim = grid[nr][nc] === 1 ? elim - 1 : elim;
      if (newElim < 0) continue;

      if (steps + 1 < dist[nr][nc][newElim]) {
        dist[nr][nc][newElim] = steps + 1;
        queue.push([nr, nc, newElim, steps + 1]);
      }
    }
  }

  return Math.min(...dist[end[0]][end[1]]) === Infinity ? -1 : Math.min(...dist[end[0]][end[1]]);
}

// ⭐ State expansion = adding dimension for constraints
// This separates SDE2 from SDE3 (recognizing state space needs)

13.1 Longest Increasing Path in Matrix (LC 329)

function longestIncreasingPath(matrix: number[][]): number {
  if (!matrix.length) return 0;

  const rows = matrix.length,
    cols = matrix[0].length;
  const memo: number[][] = Array.from({ length: rows }, () => new Array(cols).fill(0));

  function dfs(r: number, c: number): number {
    if (memo[r][c] !== 0) return memo[r][c];

    let max = 1;
    for (const [dr, dc] of [
      [0, 1],
      [0, -1],
      [1, 0],
      [-1, 0],
    ]) {
      const nr = r + dr,
        nc = c + dc;
      if (nr >= 0 && nr < rows && nc >= 0 && nc < cols && matrix[nr][nc] > matrix[r][c]) {
        max = Math.max(max, 1 + dfs(nr, nc));
      }
    }

    return (memo[r][c] = max);
  }

  let result = 0;
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      result = Math.max(result, dfs(r, c));
    }
  }

  return result;
}

// ⭐ DP on DAG:
// memo[r][c] = longest increasing path starting from (r,c)
// Implicit DAG: edge from (r,c) to neighbor if neighbor > current
// DFS + memoization = topological sort + DP

13.2 Bitmask DP (LC 864 — Keys & Rooms)

function shortestPathAllKeys(rooms: string[][]): number {
  let start = [0, 0];
  let keyCount = 0;

  // Find start position and count keys
  for (let r = 0; r < rooms.length; r++) {
    for (let c = 0; c < rooms[r].length; c++) {
      if (rooms[r][c] === '@') start = [r, c];
      if (rooms[r][c] >= 'a' && rooms[r][c] <= 'f') keyCount++;
    }
  }

  const rows = rooms.length,
    cols = rooms[0].length;
  // dist[r][c][keys_mask] = min steps to reach (r,c) with keys
  const dist = new Map<string, number>();
  dist.set(`${start[0]},${start[1]},0`, 0);

  const queue: [number, number, number, number][] = [[start[0], start[1], 0, 0]]; // [r, c, mask, steps]
  let head = 0;

  while (head < queue.length) {
    const [r, c, mask, steps] = queue[head++];

    if (mask === (1 << keyCount) - 1) return steps; // All keys

    for (const [dr, dc] of [
      [0, 1],
      [0, -1],
      [1, 0],
      [-1, 0],
    ]) {
      const nr = r + dr,
        nc = c + dc;
      if (nr < 0 || nr >= rows || nc < 0 || nc >= cols) continue;

      const cell = rooms[nr][nc];
      if (cell === '#') continue;

      let newMask = mask;
      // If it's a key, add to mask
      if (cell >= 'a' && cell <= 'f') {
        newMask |= 1 << (cell.charCodeAt(0) - 'a'.charCodeAt(0));
      }
      // If it's a lock, check if we have key
      if (cell >= 'A' && cell <= 'F') {
        if (!(mask & (1 << (cell.charCodeAt(0) - 'A'.charCodeAt(0))))) {
          continue; // Don't have key
        }
      }

      const key = `${nr},${nc},${newMask}`;
      if (!dist.has(key)) {
        dist.set(key, steps + 1);
        queue.push([nr, nc, newMask, steps + 1]);
      }
    }
  }

  return -1;
}

// ⭐ Bitmask represents which keys collected
// State = (r, c, keys_mask)
// BFS explores this 3D state space
// Exponential but manageable for small key counts

⚠️ SECTION 16: CRITICAL BUGS TO AVOID

BUG #1: Using shift() in BFS
❌ while (queue.length > 0) node = queue.shift()    // O(n) per call!
✅ while (head < queue.length) node = queue[head++]  // O(1)
   Impact: Turns O(V+E) into O(V^2+E) for sparse graphs


BUG #2: Marking visited on DEQUEUE
❌ while (queue.length > 0) {
     node = queue.shift()
     visited.add(node)                // WRONG: marks on dequeue
     for (neighbor of graph[node]) {
         queue.push(neighbor)         // Duplicate enqueues possible!
     }
   }
✅ while (queue.length > 0) {
     node = queue.shift()
     if (visited.has(node)) continue  // Skip if already processed
     visited.add(node)
   }
   Impact: Exponential duplicates in queue


BUG #3: Dijkstra without stale entry check
❌ while (heap.length) {
     [dist, node] = heap.pop()
     for ([neighbor, w] of graph[node]) {
         if (dist[node] + w < dist[neighbor]) {
             dist[neighbor] = ...
             heap.push([..., neighbor])
         }
     }
   }
✅ while (heap.length) {
     [d, node] = heap.pop()
     if (d > dist[node]) continue      // Skip stale entry
     ...
   }
   Impact: Can process same node multiple times wastefully


BUG #4: Wrong parent check in undirected cycle detection
❌ if (!visited.has(neighbor)) dfs(neighbor, node)
   else if (neighbor !== parent) return true

   Problem: If multiple edges between same nodes, breaks!

✅ Track edge ID or use Union-Find instead


BUG #5: String key collision in grid
❌ const key = `${r}${c}`   // r=1,c=23 collides with r=12,c=3!
✅ const key = `${r},${c}`  // Always use separator


BUG #6: No copy in k-stops Bellman-Ford
❌ for (i = 0; i < k; i++) {
     for ([u,v,w] of edges) {
         dist[v] = min(dist[v], dist[u] + w)  // Bleed into same iteration!
     }
   }

✅ for (i = 0; i < k; i++) {
     nextDist = [...dist]
     for ([u,v,w] of edges) {
         nextDist[v] = min(nextDist[v], dist[u] + w)
     }
     dist = nextDist
   }


BUG #7: Union-Find without path compression
   Impact: O(log n) or even O(n) operations instead of O(α(n))


BUG #8: Forgetting disconnected components
❌ dfs(graph, 0, visited)     // Only traverses from node 0!
✅ for (i = 0; i < n; i++) {
     if (!visited.has(i)) dfs(graph, i, visited)
   }
