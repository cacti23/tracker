Q1 — Leaf (or leaves) with the maximum root-to-leaf value
The post says: “return all the leaf node of the binary tree with the maximum value from root to leaf node in the path”. The wording is unclear. Most likely meaning: return every leaf whose root-to-leaf sum is the maximum, ties included. If only the best sum is wanted, drop the list and keep best.
Closest matches
LC 112 — Path Sum LC 113 — Path Sum II LC 124 — Binary Tree Maximum Path Sum (harder variant)
⏱️ TC: O(n), every node is visited once
📦 SC: O(h) for the recursion, plus the list of tied leaves
maxPathLeaves.ts
function maxPathLeaves(root: TreeNode | null): TreeNode[] {
    let best = -Infinity;
    let leaves: TreeNode[] = [];

    function dfs(node: TreeNode | null, sum: number): void {
        if (!node) return;
        sum += node.val;

        if (!node.left && !node.right) {
            if (sum > best) { best = sum; leaves = [node]; }   // new best, start a fresh list
            else if (sum === best) leaves.push(node);          // tie, keep it too
            return;
        }

        dfs(node.left, sum);
        dfs(node.right, sum);
    }

    dfs(root, 0);
    return leaves;
}
Q2 — Longest word reducible to length 1
Delete one character at a time. Every shorter word must stay in the dictionary. Find the longest word that can reach length 1 this way. Example: string → sring → sing → ing → ng → g gives length 6.
Closest matches
LC 1048 — Longest String Chain LC 720 — Longest Word in Dictionary
🪜 canReduce(word) answers one yes/no question: can this word shrink to a single letter, with every step landing on a dictionary word? One working path is enough, so return true the moment you find it. A reducible word's chain is exactly as long as the word, so there is nothing to count.
⏱️ TC: exponential in the worst case, because the same shorter word is re-tried through different delete orders
📦 SC: O(L), the recursion depth (L = word length)
longestValidWord.ts — base (no memo)
function longestValidWord(words: string[]): number {
    const wordSet = new Set(words);

    // can `word` be reduced to a single character, with every step in the dictionary?
    function canReduce(word: string): boolean {
        if (word.length === 1) return wordSet.has(word);   // 🛑 base case

        for (let i = 0; i < word.length; i++) {
            const next = word.slice(0, i) + word.slice(i + 1);

            if (wordSet.has(next) && canReduce(next)) return true;  // 🏆 one path is enough
        }

        return false;                                       // ❌ dead end
    }

    let maxLen = 0;

    for (const word of words) {
        if (canReduce(word)) maxLen = Math.max(maxLen, word.length);  // 📏 the chain length is the word's length
    }

    return maxLen;
}
⏱️ TC (follow-up): O(D · L²), each distinct word fails at most once, and each try costs an O(L) slice
📦 SC: O(D · L) for the dead set
🔁 The follow-up is a 3-line change to the code above: remember the words that already failed.
longestValidWord.ts — follow-up: skip known dead ends
    const dead = new Set<string>(); // ➕ words already proven NOT reducible

    function canReduce(word: string): boolean {
        if (word.length === 1) return wordSet.has(word);
        if (dead.has(word)) return false;           // ➕ seen this dead end before

        for (let i = 0; i < word.length; i++) {
            const next = word.slice(0, i) + word.slice(i + 1);
            if (wordSet.has(next) && canReduce(next)) return true;
        }

        dead.add(word);                                // ➕ remember it
        return false;
    }
💡 Without the dead set, the same shorter word gets reached through different delete orders and re-explored every time. This is the memo the interviewer asked for. Only failures need remembering: a success returns straight up the call stack.

function minBlocksGreedy(target: number, blocks: number[]): number {
    const sorted = [...blocks].sort((a, b) => b - a);
    let remaining = target;
    let count = 0;

    // ⚠️ Only correct when the block sizes are "canonical" (like normal coins) —
    // this is the assumption new constraints break

        while (remaining >= block) {

            count++;

    }

function minBlocksDP(target: number, blocks: number[]): number {
    const dp = new Array(target + 1).fill(Infinity);
    dp[0] = 0;

    for (let t = 1; t <= target; t++) {
        for (const block of blocks) {
            if (block <= t) dp[t] = Math.min(dp[t], dp[t - block] + 1);
        }
    }

    return dp[target] === Infinity ? -1 : dp[target];
}

class SocialGraph {
    private friends = new Map<string, Set<string>>();

    addFriendship(userA: string, userB: string): void {
        if (!this.friends.has(userA)) this.friends.set(userA, new Set());
        if (!this.friends.has(userB)) this.friends.set(userB, new Set());

        this.friends.get(userA)!.add(userB);
        this.friends.get(userB)!.add(userA);
    }

    recommendFriends(user: string): string[] {
        const direct = this.friends.get(user) ?? new Set();
        const mutualCount = new Map<string, number>();

        // Look at friends-of-friends
        for (const friend of direct) {
            for (const candidate of this.friends.get(friend) ?? []) {
                // Skip self and existing friends
                if (candidate === user || direct.has(candidate)) continue;

                mutualCount.set(candidate, (mutualCount.get(candidate) ?? 0) + 1);
            }
        }

        // Rank by mutual-friend count, highest first
        return [...mutualCount.entries()]
            .sort((a, b) => b[1] - a[1])
            .map(([id]) => id);
    }
}

🗣️ The likely data-type discussion
  1. Outer container (node → its edges): array, plain object or Map? An array adj[node] is fastest and smallest, but only when nodes are numbered 0..n−1. A plain object turns every key into a string (adj[1] and adj["1"] are the same key) and can clash with built-in names like "constructor". A Map takes any key type, has .size, and keeps insertion order. Pick array for dense numeric ids, Map otherwise.
  2. Inner container (type → neighbours): object, Map or a fixed type list? If the relationship types are a small fixed set, write them as a TypeScript type, type Rel = "friend" | "colleague" | "family", and use Record<Rel, number[]>. The compiler then catches a misspelled type. If new types can appear at runtime, use Map<string, number[]>.
  3. Neighbour list: array or Set? An array is cheapest to build and loop over, but allows duplicate edges and makes “is X a neighbour?” or “remove this edge” O(degree). A Set removes duplicates and makes those O(1), at more memory.
  4. Grouped by type vs one list of edge objects. Grouping (adj[1].friend) makes “walk only friend edges” touch only those edges. A flat list adj[1] = [{ to: 2, type: "friend" }] must filter every time, but is easier when edges gain more fields (weight, date). Say which queries matter most, then pick.
  5. Empty types: store type3: [] for every node, or leave it out and read ?? []? Leaving it out saves memory; the traversal below reads ?? [] so both work.
  6. Likely follow-ups: walk edges of a set of types (check allowed.has(type) in the loop); directed vs two-way edges (add the reverse edge, maybe with its own type, like parent/child); a path whose types must follow a pattern (BFS over (node, step in the pattern) pairs instead of nodes).

traverseByType.ts
type EdgeType = string;
type TypedGraph = Map<number, Map<EdgeType, number[]>>;

function traverseByType(graph: TypedGraph, start: number, type: EdgeType): number[] {
    const visited = new Set<number>([start]);
    const queue: number[] = [start];
    const order: number[] = [];
    let head = 0;

    while (head < queue.length) {
        const node = queue[head++];
        order.push(node);

        const neighbours = graph.get(node)?.get(type) ?? [];

        for (const next of neighbours) {
            if (visited.has(next)) continue;
            visited.add(next);
            queue.push(next);
        }
    }

    return order;
}

Example: s = "aabb" (2 a’s, 2 b’s). Cut at 1 and 3 → pieces "a" | "ab" | "b". Alice gets the middle "ab", Bob gets "a" + "b". Each has 1 a and 1 b, so return [1, 3]. For "abba" one cut is enough: "ab" | "ba" → [2].
No LeetCode match
LC 1013 — Partition Array Into Three Parts With Equal Sum (loosely related concept only)
🔑 What the cuts really mean: 3 pieces are left | middle | right. Whatever you hand out, one person always ends up with one unbroken stretch of the string (the middle, or left+middle, or middle+right), and the other gets the rest. So the question becomes: find one substring s[start..end) that holds exactly half the a’s and half the b’s. Its start and end are the cuts (skip a cut at 0 or n).
🪜 Solution: keep running counts prefixA and prefixB. A piece s[start..end) holds exactly (half a’s, half b’s) when prefix(end) − prefix(start) = (halfA, halfB). So store every prefix pair in a Map (pair → earliest index), and at each end look up the pair you need. Same trick as LC 560 Subarray Sum Equals K. An odd count of a or b means no split, so return null first.
⏱️ TC: O(n), one count pass plus one pass with O(1) Map get/set
📦 SC: O(n), the Map holds at most n + 1 prefix pairs
🧪 Checked against brute force on 20,000 random strings (every way to hand out the pieces): 0 mismatches.
✅ "aabb" → [1, 3] · "abba" → [0, 2] (only one real cut, at 2).
splitAliceBob.ts
function splitAliceBob(s: string): [number, number] | null {
    let totalA = 0;
    let totalB = 0;
    for (const ch of s) {
        if (ch === 'a') totalA++;
        else totalB++;
    }

    // 1. odd count of a or b → can't split equally
    if (totalA % 2 !== 0 || totalB % 2 !== 0) return null;

    const halfA = totalA / 2;
    const halfB = totalB / 2;
    if (s.length === 0) return [0, 0];

    // 2. find a middle piece s[start..end) that holds exactly (halfA, halfB)
    const firstSeen = new Map<string, number>(); // "a,b" prefix counts → earliest index
    firstSeen.set('0,0', 0);

    let prefixA = 0;
    let prefixB = 0;
    for (let end = 1; end <= s.length; end++) {
        if (s[end - 1] === 'a') prefixA++;
        else prefixB++;

        // piece = prefix(end) − prefix(start) must equal (halfA, halfB)
        const start = firstSeen.get(`${prefixA - halfA},${prefixB - halfB}`);
        if (start !== undefined) return [start, end]; // cuts at start and end (0 or n = no cut there)

        const key = `${prefixA},${prefixB}`;
        if (!firstSeen.has(key)) firstSeen.set(key, end);
    }

    return null;
}

canSplitAfterRemoval.ts
function canSplitAfterRemoval(nums: number[]): boolean {
    const n = nums.length;
    let total = 0;
    const first = new Map<number, number>(); // value → first index
    const last = new Map<number, number>();  // value → last index
    for (let i = 0; i < n; i++) {
        total += nums[i];
        if (!first.has(nums[i])) first.set(nums[i], i);
        last.set(nums[i], i);
    }

    let left = 0;
    for (let cut = 1; cut < n; cut++) {      // left = nums[0..cut-1], right = nums[cut..]
        left += nums[cut - 1];
        const diff = left - (total - left);  // how much heavier the left is

        // remove diff from the left: needs one other element left behind
        const li = first.get(diff);
        if (li !== undefined && li < cut && cut >= 2) return true;

        // remove -diff from the right: needs one other element left behind
        const ri = last.get(-diff);
        if (ri !== undefined && ri >= cut && n - cut >= 2) return true;
    }
    return false;
}

function luckyNumbers() {
  const isLucky = (n: number): boolean => /^[47]+$/.test(String(n));

  // BFS from 4, 7 yields lucky numbers in increasing order
  const luckyUpTo = (limit: number): number[] => {
    const out: number[] = [];
    const queue = [4, 7];
    for (let i = 0; i < queue.length; i++) {
      const cur = queue[i];
      if (cur > limit) break; // everything after is bigger
      out.push(cur);
      queue.push(cur * 10 + 4, cur * 10 + 7);
    }
    return out;
  };

  // next lucky has at most one more digit than n
  const nextLucky = (n: number): number =>
    luckyUpTo(10 ** (String(n).length + 1)).find((x) => x > n)!;

  const luckyBetween = (a: number, b: number): number[] =>
    luckyUpTo(b).filter((x) => x >= a);

  return { isLucky, nextLucky, luckyBetween };
}

Rules like “P1 is North of P2, P2 is North of P3, P3 is North of P1”. Return whether they can all be true (here: False). Directions can be N, S, E, W, NE, NW, SE, SW.

⏱️ TC: O(P + R), one pass to build the graphs and one DFS per axis (P points, R rules)

function validDirections(rules: string[]): boolean {
  // edge a -> b: a is strictly north of b (ns), or strictly east of b (ew)
  const ns = new Map<string, string[]>();
  const ew = new Map<string, string[]>();

  const addEdge = (g: Map<string, string[]>, from: string, to: string) => {
    if (!g.has(from)) g.set(from, []);
    g.get(from)!.push(to);
  };

  for (const rule of rules) {
    const [a, dir, b] = rule.split(' '); // e.g. "P1 NE P2"
    if (dir.includes('N')) addEdge(ns, a, b);
    if (dir.includes('S')) addEdge(ns, b, a);
    if (dir.includes('E')) addEdge(ew, a, b);
    if (dir.includes('W')) addEdge(ew, b, a);
  }

  // a cycle means "A north of B ... north of A": impossible
  const hasCycle = (g: Map<string, string[]>): boolean => {
    const state = new Map<string, number>(); // 1 = on current path, 2 = done
    const dfs = (u: string): boolean => {
      if (state.get(u) === 1) return true;
      if (state.get(u) === 2) return false;
      state.set(u, 1);
      for (const v of g.get(u) ?? []) if (dfs(v)) return true;
      state.set(u, 2);
      return false;
    };
    return [...g.keys()].some(dfs);
  };

  return !hasCycle(ns) && !hasCycle(ew);
}

sumOfApSubarrays.ts
function sumOfApSubarrays(nums: number[]): number {
  let total = 0;
  let len = 0; // length of the longest AP subarray ending at i
  let sum = 0; // sum of all AP subarrays ending at i

  for (let i = 0; i < nums.length; i++) {
    const continues = i >= 2 && nums[i] - nums[i - 1] === nums[i - 1] - nums[i - 2];
    if (i < 2 || continues) {
      len += 1;
      sum += nums[i] * len; // every AP ending at i-1 grows by nums[i], plus [nums[i]] alone
    } else {
      len = 2; // any pair is an AP
      sum = nums[i - 1] + 2 * nums[i]; // [prev, cur] and [cur]
    }
    total += sum;
  }
  return total;
}

removeSubtree.ts
// parent[i] = index of node i's parent, -1 for the root
function removeSubtree(parent: number[], target: number): number[] {
  const n = parent.length;
  const children: number[][] = Array.from({ length: n }, () => []);
  parent.forEach((p, i) => { if (p >= 0) children[p].push(i); });

  // mark target and everything below it
  const removed = new Array<boolean>(n).fill(false);
  const dfs = (u: number) => {
    removed[u] = true;
    for (const c of children[u]) dfs(c);
  };
  dfs(target);

  // a kept node's new index = how many kept nodes come before it
  const newIndex = new Array<number>(n).fill(-1);
  let next = 0;
  for (let i = 0; i < n; i++) if (!removed[i]) newIndex[i] = next++;

  const out: number[] = [];
  for (let i = 0; i < n; i++)
    if (!removed[i]) out.push(parent[i] === -1 ? -1 : newIndex[parent[i]]);
  return out;
}

function countHouses(start: House, max: number): number {
  let h = start;

  // max steps right visit every house at least once, so all doors end closed
  for (let i = 0; i < max; i++) {
    h.closeDoor();
    h = h.moveRight();

  // now it's the easy case: mark this house, walk until we see it again

  let count = 1;
  h = h.moveRight();
  while (!h.isDoorOpen()) {
    count++;
    h = h.moveRight();
  }
  return count;
}

interface Iter<T> { hasNext(): boolean; next(): T }

// Round 2: merge two iterators, each value returned once
function mergeDedupingIterator<T>(a: Iter<T>, b: Iter<T>): Iter<T> {
  const seen = new Set<T>();
  let buffered: T | undefined;
  let hasBuffered = false;

  // move to the next value not seen yet: drain a first, then b
  const advance = () => {
    for (const it of [a, b]) {
      while (it.hasNext()) {
        const v = it.next();
        if (!seen.has(v)) { seen.add(v); buffered = v; hasBuffered = true; return; }
      }
    }
  };

  return {
    hasNext() { if (!hasBuffered) advance(); return hasBuffered; },
    next() {
      if (!this.hasNext()) throw new Error('no more elements');
      hasBuffered = false;
      return buffered as T;
    },
  };
}

class XmlNode {
  tag: string;
  value: string;
  children: XmlNode[] = [];
  constructor(tag: string, value = '') { this.tag = tag; this.value = value; }
}

// removes the first node (pre-order) with this tag, plus everything under it
function removeFirstMatchingNode(root: XmlNode | null, tagname: string): XmlNode | null {
  if (!root) return null;
  if (root.tag === tagname) return null; // the root itself matches

  // true once a match is removed, so we stop searching
  const dfs = (node: XmlNode): boolean => {
    for (let i = 0; i < node.children.length; i++) {
      if (node.children[i].tag === tagname) { node.children.splice(i, 1); return true; }
      if (dfs(node.children[i])) return true;
    }
    return false;
  };
  dfs(root);
  return root;
}

type Card = { suit: number; rank: number };

// Round 1: highest straight flush (5 same-suit cards with consecutive ranks)
function findHighestStraightFlush(cards: Card[], suits = 4, ranks = 13): Card[] {
  // has[s][r] = true if we hold that card
  const has = Array.from({ length: suits }, () => new Array<boolean>(ranks).fill(false));
  for (const c of cards) has[c.suit][c.rank] = true;

  let bestSuit = -1;
  let bestTop = -1; // highest rank that ends a run of 5
  for (let s = 0; s < suits; s++) {
    let run = 0;
    for (let r = 0; r < ranks; r++) {
      run = has[s][r] ? run + 1 : 0;
      if (run >= 5 && r > bestTop) { bestTop = r; bestSuit = s; }
    }
  }
  if (bestTop === -1) return []; // no straight flush

  const out: Card[] = [];
  for (let r = bestTop - 4; r <= bestTop; r++) out.push({ suit: bestSuit, rank: r });
  return out;
}

// Round 4: parking strip from 0 to size; a car at x takes (x - 0.5, x + 0.5)
function createParkingLot(size = 10000) {
  const cars: number[] = []; // parked centres, kept sorted
  const EPS = 1e-9; // doubles like 12.4 aren't exact, so allow a tiny slack

  const canPark = (x: number): boolean => {
    if (x - 0.5 < -EPS || x + 0.5 > size + EPS) return false; // off the strip
    return cars.every(c => Math.abs(c - x) >= 1 - EPS); // touching is fine, overlap is not
  };

  const park = (x: number): void => {
    let i = 0;
    while (i < cars.length && cars[i] < x) i++;
    cars.splice(i, 0, x); // insert in order, like Insert Interval
  };

  // follow-up: is there a gap of length >= 1 anywhere?
  const isSpaceAvailable = (): boolean => {
    let freeFrom = 0; // end of the previous car (or the strip start)
    for (const c of cars) {
      if (c - 0.5 - freeFrom >= 1 - EPS) return true;
      freeFrom = c + 0.5;
    }
    return size - freeFrom >= 1 - EPS;
  };

  return { canPark, park, isSpaceAvailable };
}

// buf has spare cells at the end; only the first `size` cells are filled
function removeAsDoubleBs(buf: string[], size: number): number {
  // pass 1 (left to right): drop the As, count the Bs
  let kept = 0, bs = 0;
  for (let i = 0; i < size; i++) {
    if (buf[i] === 'A') continue;
    if (buf[i] === 'B') bs++;
    buf[kept++] = buf[i];
  }

  // pass 2 (right to left): write from the end so we never overwrite an unread cell
  const len = kept + bs;
  let w = len - 1;
  for (let r = kept - 1; r >= 0; r--) {
    if (buf[r] === 'B') buf[w--] = 'B'; // the extra copy
    buf[w--] = buf[r];
  }
  return len; // new filled length
}

type Cell = [number, number];

// No walls: walk columns first, then rows (matches the post's example). Length = Manhattan distance.
function shortestPath(from: Cell, to: Cell): Cell[] {
  let [r, c] = from;
  const path: Cell[] = [[r, c]];
  while (c !== to[1]) { c += c < to[1] ? 1 : -1; path.push([r, c]); }
  while (r !== to[0]) { r += r < to[0] ? 1 : -1; path.push([r, c]); }
  return path;
}

// Follow-up, if some cells are blocked (grid[r][c] === 1): BFS, remember each cell's parent.
function shortestPathWithWalls(grid: number[][], from: Cell, to: Cell): Cell[] {
  const m = grid.length, n = grid[0].length;
  const parent = new Map<number, number>(); // cell id -> previous cell id
  const id = (r: number, c: number) => r * n + c;
  const start = id(from[0], from[1]), end = id(to[0], to[1]);
  if (grid[from[0]][from[1]] === 1 || grid[to[0]][to[1]] === 1) return [];
  parent.set(start, -1);
  const queue = [start];
  for (let i = 0; i < queue.length && !parent.has(end); i++) {
    const r = Math.floor(queue[i] / n), c = queue[i] % n;
    for (const [dr, dc] of [[1, 0], [-1, 0], [0, 1], [0, -1]]) {
      const nr = r + dr, nc = c + dc;
      if (nr < 0 || nc < 0 || nr >= m || nc >= n) continue; // off the grid
      if (grid[nr][nc] === 1 || parent.has(id(nr, nc))) continue; // wall or seen
      parent.set(id(nr, nc), queue[i]);
      queue.push(id(nr, nc));
    }
  }
  if (!parent.has(end)) return []; // no path
  // walk back from the end to the start
  const path: Cell[] = [];
  for (let cur = end; cur !== -1; cur = parent.get(cur)!) path.push([Math.floor(cur / n), cur % n]);
  return path.reverse();
}
